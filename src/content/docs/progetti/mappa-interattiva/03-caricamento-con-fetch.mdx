### ⚙️ Logica Asincrona in `script.js`
Gestiamo il caricamento dell'SVG e del JSON tramite `fetch()` 
e `async/await`, collegando sia i menu `<select>`
 sia il clic diretto sugli edifici della mappa

```javascript title="script.js"
document.addEventListener("DOMContentLoaded", () => {
  caricaMappaSVG();
  caricaLocaliJSON();
});

async function caricaMappaSVG() {
  const risposta = await fetch("mappa.svg");
  document.getElementById("mappa-container").innerHTML = await risposta.text();
}

async function caricaLocaliJSON() {
  const risposta = await fetch("locali.json");
  const locali = await risposta.json();

  const selectDa = document.getElementById("select-da");
  const selectA = document.getElementById("select-a");

  locali.forEach(l => {
    selectDa.innerHTML += `<option value="${l.id}">${l.nome}</option>`;
    selectA.innerHTML += `<option value="${l.id}">${l.nome}</option>`;
  });
}

function aggiornaEvidenziazione() {
  const idDa = document.getElementById("select-da").value;
  const idA = document.getElementById("select-a").value;

  // Rimuove eventuali evidenziazioni precedenti
  document.querySelectorAll("#layer-edifici g").forEach(edificio => {
    edificio.classList.remove("selezionato-da", "selezionato-a");
  });

  // Applica le nuove classi simmetriche
  if (idDa) {
    document.getElementById(idDa)?.classList.add("selezionato-da");
  }
  if (idA) {
    document.getElementById(idA)?.classList.add("selezionato-a");
  }
}

function selezionaDaMappa(idEdificio) {
  const selectDa = document.getElementById("select-da");
  const selectA = document.getElementById("select-a");

  if (!selectDa.value) {
    selectDa.value = idEdificio;
  } else if (!selectA.value && selectDa.value !== idEdificio) {
    selectA.value = idEdificio;
  } else {
    selectDa.value = idEdificio;
    selectA.value = "";
  }

  aggiornaEvidenziazione();
}
```

<Aside type="note" title="Sintesi Concetti Chiave" icon="puzzle">
- [`mappa.svg`](https://it.wikipedia.org/wiki/Scalable_Vector_Graphics): separa la grafica della mappa dal file HTML e la rende responsive con larghezza e altezza a 100%.
- [`fetch()`](https://developer.mozilla.org/it/docs/Web/API/Fetch_API/Using_Fetch): carica file esterni come `mappa.svg` e `locali.json` in modo asincrono; approfondisci la [programmazione asincrona](https://it.wikipedia.org/wiki/Programmazione_asincrona).
- [`async` / `await`](https://developer.mozilla.org/it/docs/Web/JavaScript/Reference/Statements/async_function): aspetta il completamento delle richieste prima di usare i dati o inserire il contenuto nel DOM.
- [`locali.json`](https://www.json.org/json-it.html): contiene la lista di edifici con `id` e `nome`, utile per popolare i menu a tendina; JSON significa [JavaScript Object Notation](https://it.wikipedia.org/wiki/JavaScript_Object_Notation).
- [`innerHTML`](https://developer.mozilla.org/it/docs/Web/API/Element/innerHTML): inserisce il markup SVG caricato nel contenitore HTML della pagina, nel [DOM](https://it.wikipedia.org/wiki/Document_Object_Model).
- [`classList`](https://developer.mozilla.org/it/docs/Web/API/Element/classList): aggiunge o rimuove classi come `.selezionato-da` e `.selezionato-a` per evidenziare i locali nel [DOM](https://it.wikipedia.org/wiki/Document_Object_Model).
</Aside>
