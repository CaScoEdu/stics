## 1. The Essential `index.html` Code

<IconGroup techIconList={["vscode","seti:html"]} />
Create an `index.html` file and a `hello.txt` file in the same folder:


<Tabs>

<TabItem label="index.html" icon="seti:html" >
```html
<!DOCTYPE html>
<html lang="it">
<head>
    <meta charset="UTF-8">
    <title>Intro Async Fetch</title>
</head>
<body>

    <button onclick="loadMessage()">Load "Hello World"</button>

    <!-- A minimal container for displaying the text -->
    <div id="output">Waiting for a click...</div>

    <script>
    // 1. 'async' makes the function asynchronous
    async function loadMessage() {
        const output = document.getElementById('output');

        try {
        // 2. 'await fetch' retrieves the resource without blocking
        const response = await fetch('hello.txt');

        // 3. Check whether the file was found (e.g. 200 OK)
        if (!response.ok) {
            throw new Error('File not found!');
        }

        // 4. 'await' waits for the response body to be converted to text
        const text = await response.text();

        // 5. Add the text to the DOM (without an alert)
        output.textContent = text;

        } catch (error) {
        // Handle network issues or missing files
        output.textContent = 'Error: ' + error.message;
        }
    }
    </script>

</body>
</html>
```
</TabItem>
<TabItem label="hello.txt" icon="document">
```txt
Hello World!
```
</TabItem>
</Tabs>
